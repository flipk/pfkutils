# Convert UTF-8 to UTF-16 Little Endian (common for Windows)
iconv -f UTF-8 -t UTF-16LE input.txt -o output.txt

# Or standard UTF-16 (usually prepends a Byte Order Mark / BOM):
iconv -f UTF-8 -t UTF-16 input.txt -o output.txt
